const R=46;
function heightAt(x,z){
  const d=Math.hypot(x,z);
  let h=0.6*Math.sin(x*0.15)*Math.cos(z*0.13)+0.4*Math.sin(x*0.31+1.3)*Math.sin(z*0.27);
  h+=0.5*Math.sin((x+z)*0.09);
  h*=smooth(26,40,d);
  return h;
}
function buildTerrain(){
  const pos=[],nor=[],col=[];
  for(let x=-R;x<=R;x++)for(let z=-R;z<=R;z++){
    const d=Math.hypot(x,z);if(d>R)continue;
    const h=heightAt(x,z);
    let c;
    if(d<12)c=P.grass;
    else if(d<18)c=Math.random()<0.5?P.grass:P.grass2;
    else if(d<26)c=Math.random()<0.4?P.grass2:P.grass3;
    else if(d<33)c=Math.random()<0.5?P.grass3:P.grass2;
    else if(d<39)c=Math.random()<0.5?P.sand:P.sand2;
    else c=Math.random()<0.5?P.sand2:P.grass3;
    if(Math.random()<0.06)c=shade(c,0.9);
    pushBox(pos,nor,col,x,h-1,z,1,1.2,1);
  }
  const m=buildMesh(pos,nor,col);m.castShadow=false;scene.add(m);
}

      pushBox(pos,nor,col,bx,by,bz,1,1,1,P.trunk);}

  // canopy

  for(let dx=-cr;dx<=cr;dx++)for(let dy=-2;dy<=3;dy++)for(let dz=-cr;dz<=cr;dz++){

  g.add(buildMesh(pos,nor,col));
  g.position.y=heightAt(x,z);g.scale.setScalar(scale);

}

/* ---------- pagoda ---------- */
function buildPagoda(){
  const g=new THREE.Group();
  const pos=[],nor=[],col=[];
  const add=(x,y,z,sx,sy,sz,c)=>pushBox(pos,nor,col,x,y,z,sx,sy,sz,c);

  // stone platform
  for(let x=-11;x<=11;x++)for(let z=-11;z<=11;z++)add(x,-1,z,1,1,1,Math.random()<0.5?P.stone:P.stoneDark);
  for(let x=-9;x<=9;x++)for(let z=-9;z<=9;z++)add(x,0,z,1,1,1,Math.random()<0.5?P.stoneLight:P.stone);
  // steps (front +z)
  for(let s=0;s<3;s++)for(let x=-3;x<=3;x++)add(x,-1+s,11+s,1,1,1,P.stone);

  // tiers
  const tiers=[{hw:6,h:4},{hw:5,h:4},{hw:4,h:4},{hw:3,h:3},{hw:2,h:3}];
  let y=1;
  tiers.forEach((t,i)=>{
    const hw=t.hw;
    // body
    for(let x=-hw;x<=hw;x++)for(let z=-hw;z<=hw;z++)for(let yy=0;yy<t.h;yy++){
      const edge=(Math.abs(x)===hw||Math.abs(z)===hw);
      let c;
      if(edge&&yy===0)c=P.woodDark;
      else if(edge&&yy===t.h-1)c=P.woodDark;
      else if(edge)c=(Math.abs(x)===hw&&Math.abs(z)===hw)?P.red:(Math.random()<0.5?P.wall:P.wallDark);
      else c=P.wood;
      add(x,y+yy,z,1,1,1,c);
    }
    // corner posts
    for(let yy=0;yy<t.h;yy++){
      add(-hw-1,y+yy,-hw-1,1,1,1,P.red);add(hw+1,y+yy,-hw-1,1,1,1,P.red);
      add(-hw-1,y+yy,hw+1,1,1,1,P.red);add(hw+1,y+yy,hw+1,1,1,1,P.red);
    }
    // windows on faces
    for(let s=-hw+1;s<=hw-1;s+=2){
      add(s,y+1,hw+1,1,2,1,P.window);add(s,y+1,-hw-1,1,2,1,P.window);
      add(hw+1,y+1,s,1,2,1,P.window);add(-hw-1,y+1,s,1,2,1,P.window);
    }
    // shoji door front
    if(i===0){for(let x=-1;x<=1;x++)add(x,y+1,hw+1,1,3,1,P.shoji);
      add(-1,y+1,hw+1,1,3,1,P.shojiFrame);add(1,y+1,hw+1,1,3,1,P.shojiFrame);}
    // gold band
    for(let x=-hw-1;x<=hw+1;x++)for(let z=-hw-1;z<=hw+1;z++)
      if(Math.abs(x)===hw+1||Math.abs(z)===hw+1)add(x,y+t.h,z,1,1,1,P.gold);
    // roof
    const rw=hw+3;
    for(let x=-rw;x<=rw;x++)for(let z=-rw;z<=rw;z++){
      const e=Math.max(Math.abs(x),Math.abs(z));
      if(e>rw)continue;
      const up=rw-e;
      const c=e===rw?P.roofTileDark:(e===rw-1?P.roofTile:(e===rw-2?P.roofTileLight:P.roofTile));
      add(x,y+t.h+1+up,z,1,1,1,c);
    }
    // gold ridge
    for(let x=-rw;x<=rw;x++)add(x,y+t.h+1+rw,0,1,1,1,P.roofGold);
    for(let z=-rw;z<=rw;z++)add(0,y+t.h+1+rw,z,1,1,1,P.roofGold);
    // upturned corners
    for(const sx of[-1,1])for(const sz of[-1,1]){
      add(sx*rw,y+t.h+1+rw+1,sz*rw,1,1,1,P.roofGold);
      add(sx*(rw-1),y+t.h+1+rw,sz*(rw-1),1,1,1,P.roofGoldDark);
    }
    y+=t.h+1+rw+1;
  });

  // spire
  for(let yy=0;yy<6;yy++)add(0,y+yy,0,1,1,1,P.gold);
  for(let r=2;r>=1;r--)for(let x=-r;x<=r;x++)for(let z=-r;z<=r;z++)
    if(Math.max(Math.abs(x),Math.abs(z))===r)add(x,y+2+(2-r),z,1,1,1,P.goldDark);
  add(0,y+6,0,1,2,1,P.gold);
  add(0,y+8,0,1,1,1,P.gold);

  const mesh=buildMesh(pos,nor,col);
  g.add(mesh);
  // interior warm glow
  const glow=new THREE.PointLight(0xffcf8a,10,26,2);glow.position.set(0,7,0);g.add(glow);
  scene.add(g);
}

A few things worth knowing:

- **Everything is one file** — just save as `.html` and open. It pulls Three.js from a CDN via an import map, so it needs an internet connection on first load.
- **Interactions:** drag to orbit, scroll to zoom, right-drag to pan. The panel toggles **Dawn / Day / Dusk** (the sun, sky fog, water tint, and exposure all cross-fade smoothly), plus petals, auto-orbit, and the water reflection.
- **Scene contents:** a five-tier pagoda with upturned gold-ridged roofs, glowing shoji windows, a koi pond with a real mirror reflection and swimming koi, a torii gate, an arched bridge, stone lanterns with point-light glows, cherry/maple/green trees, bamboo groves, rock clusters, flower beds, and ~650 drifting petals.
- **Performance:** the whole garden is merged into a handful of vertex-colored meshes (no per-voxel draw calls), so it stays smooth even with soft shadows. If your GPU struggles, drop `renderer.setPixelRatio(Math.min(devicePixelRatio,2))` to `1`.
