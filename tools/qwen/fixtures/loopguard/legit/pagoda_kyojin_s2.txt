/* ============================ helpers ============================ */
const clamp=(v,a,b)=>Math.max(a,Math.min(b,v));
const lerp=(a,b,t)=>a+(b-a)*t;
const lerp3=(a,b,t)=>[lerp(a[0],b[0],t),lerp(a[1],b[1],t),lerp(a[2],b[2],t)];
const lerpC=(a,b,t)=>new THREE.Color(a).lerp(new THREE.Color(b),t);
const easeIO=t=>t<.5?4*t*t*t:1-Math.pow(-2*t+2,3)/2;
const smooth=(a,b,x)=>{const t=clamp((x-a)/(b-a),0,1);return t*t*(3-2*t);};
const hash=(x,y,z)=>{let h=(x*374761393+y*668265263+z*1274126177)>>>0;h=(h^(h>>13))>>>0;h=(h*1274126177)>>>0;return (h^(h>>16))>>>0;};
const rnd=(x,y,z)=>hash(x,y,z)/4294967295;
const pick=(arr,x,y,z)=>arr[Math.floor(rnd(x,y,z)*arr.length)%arr.length];
const shade=(hex,f)=>{const c=new THREE.Color(hex);c.multiplyScalar(f);return c;};
const jitter=(hex,a,x,y,z)=>shade(hex,1+(rnd(x,y,z)*2-1)*a);

/* ============================ palette ============================ */

const voxelGeo=new THREE.BoxGeometry(1,1,1);

function build(name,voxels,opts={}){
  const g=new THREE.Group(); g.name=name;
  const buckets=new Map();
  for(const v of voxels){
    const key=v.color.getHex();
    if(!buckets.has(key))buckets.set(key,[]);
    buckets.get(key).push(v);
  }
  for(const [hex,list] of buckets){
    const mat=new THREE.MeshLambertMaterial({color:hex});
    const im=new THREE.InstancedMesh(voxelGeo,mat,list.length);
    im.castShadow=opts.cast!==false; im.receiveShadow=opts.receive!==false;
    const m=new THREE.Matrix4();
    list.forEach((v,i)=>{m.makeTranslation(v.x,v.y,v.z); im.setMatrixAt(i,m);});
    im.instanceMatrix.needsUpdate=true;
    im.userData={name,voxels:list};
    g.add(im);
  }
  g.userData={name,voxels};
  group.add(g);
  if(opts.pick!==false)pickables.push(g);
  return g;
}

/* ============================ builders ============================ */
function buildGround(){
  const R=62, vox=[];
  for(let x=-R;x<=R;x++)for(let z=-R;z<=R;z++){
    const d=Math.hypot(x,z); if(d>R)continue;
    const n=rnd(x,7,z);
    let c;
    if(d>R-2.5) c=jitter(P.stoneDark,0.05,x,0,z);
    else if(n<0.16) c=jitter(P.mossDark,0.06,x,1,z);
    else if(n<0.5) c=jitter(P.mossMid,0.06,x,2,z);
    else if(n<0.82) c=jitter(P.mossLight,0.06,x,3,z);
    else c=jitter(P.grassC,0.06,x,4,z);
    vox.push(V(x,-1,z,c));
  }
  build('ground',vox,{cast:false});
}

function buildPath(){
  const vox=[];
  const line=(x,z)=>{const n=rnd(x,9,z);
    if(n<0.25)return jitter(P.stoneDark,0.05,x,5,z);
    if(n<0.6)return jitter(P.stoneMid,0.05,x,6,z);
    return jitter(P.stoneLight,0.05,x,7,z);};
  for(let z=30;z>=12;z--){const w=2+Math.round(Math.sin(z*0.4)*0.8);
    for(let x=-w;x<=w;x++)vox.push(V(x,0,z,line(x,z)));}
  for(let x=-18;x<=18;x++)for(let z=6;z<=12;z++)vox.push(V(x,0,z,line(x,z)));
  for(let x=-24;x<=24;x++)for(let z=2;z<=6;z++)vox.push(V(x,0,z,line(x,z)));
  build('path',vox,{cast:false});
}

function buildPond(){
  const cx=-30,cz=24,rx=15,rz=11,vox=[];
  for(let x=cx-rx-2;x<=cx+rx+2;x++)for(let z=cz-rz-2;z<=cz+rz+2;z++){
    const e=((x-cx)/rx)**2+((z-cz)/rz)**2;
    if(e<=1){
      const n=rnd(x,11,z);
      let c = e<0.45?jitter(P.waterDeep,0.05,x,0,z) : e<0.78?jitter(P.waterMid,0.05,x,1,z) : jitter(P.waterShallow,0.05,x,2,z);
      vox.push(V(x,-1,z,c));
    } else if(e<=1.28){
      const n=rnd(x,12,z);
      vox.push(V(x,0,z, n<0.4?jitter(P.stoneDark,0.05,x,3,z): n<0.75?jitter(P.stoneMid,0.05,x,4,z):jitter(P.stoneLight,0.05,x,5,z)));
    }
  }
  build('pond',vox,{cast:false});
}

function buildBridge(){
  const vox=[];
  for(let x=-40;x<=-20;x++){
    const t=(x+40)/20, y=Math.round(2+Math.sin(t*Math.PI)*3);
    for(let z=22;z<=25;z++)vox.push(V(x,y,z,jitter(P.woodMid,0.05,x,0,z)));
    for(let z of [22,25]){vox.push(V(x,y+1,z,jitter(P.woodDark,0.05,x,1,z)));
      if((x+z)%3===0)vox.push(V(x,y+2,z,jitter(P.woodDark,0.05,x,2,z)));}
  }
  build('bridge',vox);
}

function buildPagoda(){
  const vox=[];
  const add=(x,y,z,c)=>vox.push(V(x,y,z,c));
  const ring=(cx,cz,r,y,c)=>{for(let x=-r;x<=r;x++)for(let z=-r;z<=r;z++){const d=Math.hypot(x-cx,z-cz);
    if(d<=r&&d>r-1.15)add(cx+x,y,cz+z,c);}};
  const disc=(cx,cz,r,y,c)=>{for(let x=-r;x<=r;x++)for(let z=-r;z<=r;z++)if(Math.hypot(x-cx,z-cz)<=r)add(cx+x,y,cz+z,c);};
  const box=(x0,x1,y0,y1,z0,z1,c)=>{for(let x=x0;x<=x1;x++)for(let y=y0;y<=y1;y++)for(let z=z0;z<=z1;z++)add(x,y,z,c);};

  // stone plinth
  box(-15,15,0,0,-15,15,jitter(P.stoneMid,0.04,0,0,0));
  box(-13,13,1,1,-13,13,jitter(P.stoneLight,0.04,0,1,0));
  box(-11,11,2,2,-11,11,jitter(P.stoneMid,0.04,0,2,0));
  // steps
  for(let s=0;s<3;s++){const w=15-s*2; for(let x=-w;x<=w;x++)add(x,s,15+s,jitter(P.stoneLight,0.04,x,3,s));}

  // 5 tiers
  const levels=[{r:10,h:6},{r:9,h:5},{r:8,h:5},{r:7,h:4},{r:6,h:4}];
  let y=3;
  levels.forEach((L,i)=>{
    const r=L.r, h=L.h;
    // walls
    for(let x=-r;x<=r;x++)for(let z=-r;z<=r;z++){const d=Math.hypot(x,z);
      if(d<=r&&d>r-1.15){
        let c = (x===0||z===0)?jitter(P.woodDark,0.04,x,y,z):jitter(P.wall,0.03,x,y,z);

A few things worth knowing:

- **The garden** — a five-tier vermilion-and-gold pagoda on a stepped stone plinth, a torii gate, a red arched bridge over a koi pond (with four swimming koi), stone lanterns, cherry/maple/pine/bamboo trees, rock clusters, flower beds, and a winding stone path — all built from per-voxel instanced meshes, so it stays fast.
- **Living details** — drifting cherry petals, swimming koi, soft shadows, and a slow auto-orbit.
- **Controls** — drag to orbit, scroll to zoom, click any structure to fly the camera to it. The panel toggles auto-orbit and petals, resets/top-down views, and the **Time of day** slider relights the whole scene from dawn to dusk (sky, sun, fog, exposure all shift together). **Petal density** scales the falling-blossom count.
