function buildGround(){
  const R=30;
  for(let x=-R;x<=R;x++)for(let z=-R;z<=R;z++){
    const d=Math.hypot(x,z); if(d>R) continue;
    const n=hash(x*3+1,z*7+2);
    let col = n<0.14?C.grassD : n<0.30?C.grassL : C.grass;
    if(d>R-1.5 && n<0.5) col=C.grassD;
    setVoxel(x,0,z,col);
  }
}
function buildPond(){
  const cx=13,cz=11,rx=7,rz=5;
  for(let x=cx-rx-2;x<=cx+rx+2;x++)for(let z=cz-rz-2;z<=cz+rz+2;z++){
    const e=(x-cx)/(rx+1.6),f=(z-cz)/(rz+1.6),d=e*e+f*f;
    if(d<=1){
      const n=hash(x*5+9,z*11+3);
      setVoxel(x,0,z, n<0.2?C.sandD : n<0.5?C.sand : C.sand);
      setVoxel(x,1,z, n<0.25?C.waterD : n<0.55?C.water : C.waterL);
    } else if(d<=1.7){
      const n=hash(x*13+5,z*17+7);
      setVoxel(x,0,z, n<0.3?C.gravelD : n<0.6?C.gravel : C.stone);
    }
  }
  // lily pads
  const pads=[[10,9],[15,13],[12,14],[16,9],[11,12]];
  for(const [px,pz] of pads){
    for(let x=px-1;x<=px+1;x++)for(let z=pz-1;z<=pz+1;z++){
      if(Math.hypot(x-px,z-pz)<=1.2 && hasVoxel(x,1,z)) setVoxel(x,2,z, hash(x,z)<0.5?C.leaf:C.leafL);
    }
    if(hasVoxel(px,2,pz)) setVoxel(px,3,pz, pick([C.sakura,C.sakuraL,C.white]));
  }
}
function buildPath(){
